namespace {
constexpr size_t N = size_t(Stage::Count);
constexpr const char *NAMES[] = {"database_wait",
                                 "admission",
                                 "publication_wait",
                                 "payload_guard",
                                 "pins_wait",
                                 "begin",
                                 "commit",
                                 "statement_acquire",
                                 "statement_step",
                                 "statement_release",
                                 "exec",
                                 "writeback_error",
                                 "pause_enter",
                                 "pause_exit",
                                 "publication",
                                 "notify",
                                 "checkpoint",
                                 "wal_hook",
                                 "pread",
                                 "pwrite",
                                 "fsync",
                                 "fdatasync",
                                 "hint",
                                 "presync", "restart"};
static_assert(sizeof(NAMES) / sizeof(*NAMES) == N);
struct Row {
	const char *name;
	long long begin, end = 0, cpu_begin, cpu_end = 0;
	int64_t offset, bytes;
	uint64_t steps = 0;
	std::array<int, 4> wal {-1, -1, -1, 0};
	int committed_frames = -1;
	std::array<long long, 7> decision {-1, -1, -1, -1, -1, -1, -1};
	std::array<uint64_t, N> counts {}, ns {};
};
struct Thread {
	long tid = 0;
	std::vector<Row> rows;
	struct Sync {
		int64_t request, size;
		const char *kind;
		const char *site;
		const char *operation;
		int fd, result, error;
		long long begin, end, cpu, probe;
	};
	std::vector<Sync> syncs;
	std::vector<std::array<long long, 7>> checkpoints;
};
Thread threads[64];
std::atomic<unsigned> next {0};
thread_local int index = -1;
thread_local Row *active = nullptr;
thread_local const char *sync_site = "unmarked";
std::chrono::steady_clock::time_point origin;
long long epoch;
std::string output;
std::string database_path;
Thread &CurrentThread() {
	if (index < 0) {
		index = int(next.fetch_add(1));
		if (index >= 64)
			std::abort();
		threads[index].tid = syscall(SYS_gettid);
		threads[index].rows.reserve(65536);
		threads[index].syncs.reserve(65536);
		threads[index].checkpoints.reserve(131072);
	}
	return threads[index];
}
long long Cpu() {
	int saved = errno;
	struct timespec time {};
	if (clock_gettime(CLOCK_THREAD_CPUTIME_ID, &time))
		std::abort();
	auto result = int64_t(time.tv_sec) * 1000000000 + time.tv_nsec;
	errno = saved;
	return result;
}
void Report() {
	std::ofstream file(output);
	file << "{\"origin_epoch_ns\":" << epoch << ",\"stages\":[";
	for (size_t i = 0; i < N; ++i) {
		if (i)
			file << ',';
		file << std::quoted(NAMES[i]);
	}
	file << "],\"columns\":[\"name\",\"begin_ns\",\"end_ns\",\"cpu_ns\",\"offset\",\"bytes\",\"vm_steps\",\"counts\","
	        "\"stage_ns\",\"last_checkpoint\",\"committed_frames\",\"decision\"],"
	        "\"sync_columns\":[\"request_index\",\"file_size\",\"kind\",\"site\",\"operation\",\"fd\",\"result\","
	        "\"errno\","
	        "\"begin_ns\",\"end_ns\",\"cpu_ns\",\"probe_ns\"],\"threads\":[";
	for (unsigned i = 0; i < next; ++i) {
		if (i)
			file << ',';
		file << "{\"tid\":" << threads[i].tid << ",\"rows\":[";
		bool first = true;
		for (const auto &row : threads[i].rows) {
			if (!first)
				file << ',';
			first = false;
			file << '[' << std::quoted(row.name) << ',' << row.begin << ',' << row.end << ','
			     << row.cpu_end - row.cpu_begin << ',' << row.offset << ',' << row.bytes << ',' << row.steps;
			for (auto values : {&row.counts, &row.ns}) {
				file << ",[";
				for (size_t j = 0; j < N; ++j) {
					if (j)
						file << ',';
					file << (*values)[j];
				}
				file << ']';
			}
			file << ",[" << row.wal[0] << ',' << row.wal[1] << ',' << row.wal[2] << ',' << row.wal[3] << "],"
			     << row.committed_frames << ",[";
            for (size_t j = 0; j < row.decision.size(); ++j) {
                if (j) file << ',';
                file << row.decision[j];
            }
            file << "]]";
		}
		file << "],\"syncs\":[";
		first = true;
		for (const auto &call : threads[i].syncs) {
			if (!first)
				file << ',';
			first = false;
			file << '[' << call.request << ',' << call.size << ',' << std::quoted(call.kind) << ','
			     << std::quoted(call.site) << ',' << std::quoted(call.operation) << ',' << call.fd << ',' << call.result
			     << ',' << call.error << ',' << call.begin << ',' << call.end << ',' << call.cpu << ',' << call.probe
			     << ']';
		}
		file << "],\"checkpoints\":[";
        first = true;
        for (const auto &call : threads[i].checkpoints) {
            if (!first) file << ',';
            first = false;
            file << '[';
            for (size_t j = 0; j < call.size(); ++j) {
                if (j) file << ',';
                file << call[j];
            }
            file << ']';
        }
        file << "]}";
	}
	file << "]}\n";
	if (!file)
		std::abort();
}
} // namespace
long long Now() {
	return std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::steady_clock::now() - origin).count();
}
void Add(Stage stage, long long begin) {
	if (!active)
		return;
	auto end = Now();
	++active->counts[size_t(stage)];
	active->ns[size_t(stage)] += end - begin;
}
void Steps(int value) {
	if (active && value >= 0)
		active->steps += value;
}
void Wal(int mode, int frames, int copied, int rc) {
	if (active)
		active->wal = {mode, frames, copied, rc};
}
void Decision(bool pending, int frames, int copied, bool retry, bool deferred, long long remaining) {
    if (active) active->decision = {Now(), pending, frames, copied, retry, deferred, remaining};
}
void Checkpoint(int mode, int frames, int copied, int code, long long begin) {
    auto &thread = CurrentThread();
    if (thread.checkpoints.size() == thread.checkpoints.capacity()) std::abort();
    thread.checkpoints.push_back({active ? int64_t(active - thread.rows.data()) : -1, mode, frames, copied, code, begin, Now()});
}
void CommitFrames(int frames) {
	if (active)
		active->committed_frames = frames;
}
SyncSite::SyncSite(const char *name) : previous(sync_site) {
	sync_site = name;
}
SyncSite::~SyncSite() {
	sync_site = previous;
}
Scope::Scope(const char *name, int64_t offset, int64_t bytes) : previous(active) {
	auto &rows = CurrentThread().rows;
	if (rows.size() == rows.capacity())
		std::abort();
	rows.push_back({name, Now(), 0, Cpu(), 0, offset, bytes});
	record = active = &rows.back();
}
Scope::~Scope() {
	auto &row = *static_cast<Row *>(record);
	row.cpu_end = Cpu();
	row.end = Now();
	active = static_cast<Row *>(previous);
}
void Install(const char *database) {
	output = std::string(database) + ".request-costs.json";
	database_path = std::filesystem::weakly_canonical(database).string();
	origin = std::chrono::steady_clock::now();
	epoch = std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::system_clock::now().time_since_epoch())
	            .count();
	std::atexit(Report);
}
int Sync(int fd, const char *operation, int (*function)(int), Stage stage) {
	int initial_errno = errno;
	auto &thread = CurrentThread();
	auto probe_begin = Now();
	std::array<char, 64> descriptor {};
	std::snprintf(descriptor.data(), descriptor.size(), "/proc/self/fd/%d", fd);
	std::array<char, 4096> target {};
	auto length = readlink(descriptor.data(), target.data(), target.size() - 1);
	const char *kind = "other";
	struct stat stat {};
	auto size = fstat(fd, &stat) ? int64_t(-1) : int64_t(stat.st_size);
	if (length >= 0) {
		// Descriptor reuse and different open-file descriptions are handled by
		// looking up the current descriptor, with no shared registry or cache.
		constexpr const char *deleted = " (deleted)";
		if (size_t(length) >= std::strlen(deleted) &&
		    !std::strcmp(target.data() + length - std::strlen(deleted), deleted))
			target[size_t(length) - std::strlen(deleted)] = 0;
		auto prefix = database_path.size();
		if (!std::strncmp(target.data(), database_path.c_str(), prefix)) {
			auto suffix = target.data() + prefix;
			if (!std::strcmp(suffix, ".payload"))
				kind = "payload";
			else if (!std::strcmp(suffix, "-wal"))
				kind = "wal";
			else if (!std::strcmp(suffix, ".stage-fence"))
				kind = "fence";
			else if (!*suffix)
				kind = "database";
		}
		if (size >= 0 && S_ISDIR(stat.st_mode))
			kind = "directory";
	}
	auto probe = Now() - probe_begin;
	errno = initial_errno;
	auto cpu_begin = Cpu();
	auto begin = Now();
	auto result = function(fd);
	int saved_errno = errno;
	auto end = Now();
	auto cpu = Cpu() - cpu_begin;
	if (active) {
		++active->counts[size_t(stage)];
		active->ns[size_t(stage)] += end - begin;
	}
	if (thread.syncs.size() == thread.syncs.capacity())
		std::abort();
	thread.syncs.push_back({active ? int64_t(active - thread.rows.data()) : -1, size, kind, sync_site, operation, fd,
	                        result, result ? saved_errno : 0, begin, end, cpu, probe});
	errno = saved_errno;
	return result;
}
} // namespace request_trace

// Wrappers preserve the original syscall and errno. Stage times are inclusive:
// SQLite step/commit/admission can contain the measured I/O/checkpoint time.
extern "C" int __real_fsync(int);
extern "C" int __real_fdatasync(int);
extern "C" int __wrap_fsync(int fd) {
	return request_trace::Sync(fd, "fsync", __real_fsync, request_trace::Stage::Fsync);
}
extern "C" int __wrap_fdatasync(int fd) {
	return request_trace::Sync(fd, "fdatasync", __real_fdatasync, request_trace::Stage::Fdatasync);
}
extern "C" ssize_t __real_pread(int, void *, size_t, off_t);
extern "C" ssize_t __real_pwrite(int, const void *, size_t, off_t);
extern "C" ssize_t __real_pread64(int, void *, size_t, off64_t);
extern "C" ssize_t __real_pwrite64(int, const void *, size_t, off64_t);
extern "C" ssize_t __wrap_pread(int fd, void *data, size_t bytes, off_t offset) {
	request_trace::Span span(request_trace::Stage::Pread);
	return __real_pread(fd, data, bytes, offset);
}
extern "C" ssize_t __wrap_pwrite(int fd, const void *data, size_t bytes, off_t offset) {
	request_trace::Span span(request_trace::Stage::Pwrite);
	return __real_pwrite(fd, data, bytes, offset);
}
extern "C" ssize_t __wrap_pread64(int fd, void *data, size_t bytes, off64_t offset) {
	request_trace::Span span(request_trace::Stage::Pread);
	return __real_pread64(fd, data, bytes, offset);
}
extern "C" ssize_t __wrap_pwrite64(int fd, const void *data, size_t bytes, off64_t offset) {
	request_trace::Span span(request_trace::Stage::Pwrite);
	return __real_pwrite64(fd, data, bytes, offset);
}
extern "C" int __real_sqlite3_wal_checkpoint_v2(sqlite3 *, const char *, int, int *, int *);
extern "C" int __wrap_sqlite3_wal_checkpoint_v2(sqlite3 *db, const char *name, int mode, int *frames, int *copied) {
	request_trace::Span span(request_trace::Stage::Checkpoint);
	auto begin = request_trace::Now();
	auto rc = __real_sqlite3_wal_checkpoint_v2(db, name, mode, frames, copied);
	request_trace::Checkpoint(mode, frames ? *frames : -1, copied ? *copied : -1, rc, begin);
	request_trace::Wal(mode, frames ? *frames : -1, copied ? *copied : -1, rc);
	return rc;
}
extern "C" int __real_sync_file_range(int, off64_t, off64_t, unsigned);
extern "C" int __wrap_sync_file_range(int fd, off64_t offset, off64_t bytes, unsigned flags) {
	request_trace::Scope request("hint", offset, bytes);
	request_trace::Span span(request_trace::Stage::Hint);
	return __real_sync_file_range(fd, offset, bytes, flags);
}
