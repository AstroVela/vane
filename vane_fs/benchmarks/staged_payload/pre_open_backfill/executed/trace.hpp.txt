#pragma once

#include <cstdint>
#include <mutex>

// Diagnostic copies only. Short spans accumulate in the active request.
// Sync wrappers separately inspect /proc/self/fd and retain individual calls.
namespace request_trace {
enum class Stage {
	DatabaseWait,
	Admission,
	PublicationWait,
	PayloadGuard,
	PinsWait,
	Begin,
	Commit,
	StatementAcquire,
	StatementStep,
	StatementRelease,
	Exec,
	WritebackError,
	PauseEnter,
	PauseExit,
	Publication,
	Notify,
	Checkpoint,
	WalHook,
	Pread,
	Pwrite,
	Fsync,
	Fdatasync,
	Hint,
	Presync,
	Restart,
	Count
};
void Install(const char *database);
long long Now();
void Add(Stage stage, long long begin);
void Steps(int value);
void Wal(int mode, int frames, int copied, int rc);
void CommitFrames(int frames);
void Decision(bool pending, int frames, int copied, bool retry, bool deferred, long long remaining);
void Checkpoint(int mode, int frames, int copied, int code, long long begin);
class SyncSite {
public:
	explicit SyncSite(const char *name);
	~SyncSite();
	SyncSite(const SyncSite &) = delete;
	SyncSite &operator=(const SyncSite &) = delete;

private:
	const char *previous;
};
class Scope {
public:
	explicit Scope(const char *name, int64_t offset = -1, int64_t bytes = -1);
	~Scope();
	Scope(const Scope &) = delete;
	Scope &operator=(const Scope &) = delete;

private:
	void *previous;
	void *record;
};
class Span {
public:
	explicit Span(Stage stage) : stage(stage), begin(Now()) {
	}
	~Span() {
		Add(stage, begin);
	}

private:
	Stage stage;
	long long begin;
};
inline std::unique_lock<std::mutex> Lock(Stage stage, std::mutex &mutex) {
	Span span(stage);
	return std::unique_lock<std::mutex>(mutex);
}
} // namespace request_trace
